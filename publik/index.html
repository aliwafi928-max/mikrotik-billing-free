Notifikasi Kendala Otomatis

Kirim laporan WA otomatis ke Admin/Teknisi saat terdeteksi user LOS, Offline, atau sering DC.

// --- NOTIFIKASI KENDALA OTOMATIS --- let monitorTimer = null; function openNotifModal() { const cfg = JSON.parse(localStorage.getItem('notif_cfg') || '{}'); document.getElementById('adminWaInput').value = cfg.adminWa || ''; document.getElementById('notifInterval').value = cfg.interval || '180000'; document.getElementById('notifEnabled').checked = Boolean(cfg.enabled); document.getElementById('notifModal').classList.remove('hidden'); } function saveNotifSettings() { const cfg = { adminWa: document.getElementById('adminWaInput').value.trim(), interval: Number(document.getElementById('notifInterval').value) || 180000, enabled: document.getElementById('notifEnabled').checked }; if (cfg.enabled && !cfg.adminWa) return alert('Isi Nomor WA Admin / Teknisi terlebih dahulu.'); localStorage.setItem('notif_cfg', JSON.stringify(cfg)); closeModal('notifModal'); startErrorMonitor(); alert(cfg.enabled ? 'Notifikasi kendala otomatis AKTIF.' : 'Notifikasi kendala otomatis DIMATIKAN.'); } function startErrorMonitor() { if (monitorTimer) clearInterval(monitorTimer); const cfg = JSON.parse(localStorage.getItem('notif_cfg') || '{}'); const btn = document.getElementById('btnAutoNotif'); if (cfg.enabled && cfg.adminWa) { btn.textContent = 'Notif Kendala: ON'; btn.className = 'bg-orange-600 hover:bg-orange-500 px-3 py-2 rounded-lg text-sm font-medium'; monitorTimer = setInterval(async () => { await loadData(); await checkAndNotifyNewErrors(); }, cfg.interval || 180000); } else { btn.textContent = 'Notif Kendala: OFF'; btn.className = 'bg-slate-800 hover:bg-slate-700 px-3 py-2 rounded-lg text-sm font-medium'; } } async function checkAndNotifyNewErrors() { const cfg = JSON.parse(localStorage.getItem('notif_cfg') || '{}'); const mtCfg = JSON.parse(localStorage.getItem('mt_cfg') || '{}'); if (!cfg.enabled || !cfg.adminWa || !mtCfg.waToken) return; const notified = JSON.parse(localStorage.getItem('notified_errors') || '{}'); const newErrors = []; // Cek user yang mengalami kendala baru for (const u of allUsers) { const errText = u.autoError || u.errorNote || ''; if (errText) { if (notified[u.name] !== errText) { newErrors.push(`• *${u.name}* (${u.profile}): ${errText}`); notified[u.name] = errText; } } else if (notified[u.name]) { // Hapus dari daftar jika sudah kembali normal delete notified[u.name]; } } localStorage.setItem('notified_errors', JSON.stringify(notified)); if (newErrors.length === 0) return; const msg = `*PERINGATAN KENDALA PPPoE*\n\nTerdeteksi ${newErrors.length} gangguan baru:\n${newErrors.join('\n')}`; await fetch('/api/wa/send', { method: 'POST', headers: getHeaders(), body: JSON.stringify({ targets: [{ name: 'Admin', phone: cfg.adminWa, message: msg }] }) }); } async function sendErrorReportNow() { const adminWa = document.getElementById('adminWaInput').value.trim(); if (!adminWa) return alert('Isi Nomor WA Admin / Teknisi terlebih dahulu.'); const list = allUsers .filter(u => u.autoError || u.errorNote) .map(u => `• *${u.name}* (${u.profile}): ${u.autoError || u.errorNote}`); if (list.length === 0) return alert('Semua user dalam keadaan normal (tidak ada kendala).'); const msg = `*LAPORAN KENDALA PPPoE SAAT INI*\nTotal: ${list.length} user bermasalah\n\n${list.join('\n')}`; const mtCfg = JSON.parse(localStorage.getItem('mt_cfg') || '{}'); if (mtCfg.waToken) { const res = await fetch('/api/wa/send', { method: 'POST', headers: getHeaders(), body: JSON.stringify({ targets: [{ name: 'Admin', phone: adminWa, message: msg }] }) }); const data = await res.json(); alert(data.message); } else { const cleanPhone = adminWa.replace(/[^0-9]/g, ''); window.open(`https://wa.me/${cleanPhone}?text=${encodeURIComponent(msg)}`, '_blank'); } } // Jalankan monitor otomatis saat berhasil login const origLoadData = loadData; loadData = async function() { await origLoadData(); if (!monitorTimer) startErrorMonitor(); };
